3)#Write a Java script to find the GCD of a given TWO numbers.

const prompt = require("prompt-sync")();
function findGCD(a, b) {
    while (b !== 0) {
        let temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}
const num1 = parseInt(prompt("Enter first number: "));
const num2 = parseInt(prompt("Enter second number: "));
const result = findGCD(num1, num2);
console.log(`The GCD of ${num1} and ${num2} is: ${result}`);

async function fileOperations() {
    try {
        // Write
        await fs.writeFile('test.txt', 'Initial content');
        console.log('File written');

        // Read
        const data = await fs.readFile('test.txt', 'utf8');
        console.log('Read content:', data);

        // Append
        await fs.appendFile('test.txt', '\nAppended content');
        console.log('Content appended');

        // Rename
        await fs.rename('test.txt', 'new_name.txt');

        // Delete

        console.error('Error:', err);
    }
}

const http = require('http');
const url = require('url');
const server = http.createServer((req, res) => {
    const queryObject = url.parse(req.url, true).query;
    res.writeHead(200, { 'Content-Type': 'text/html' });
    if (queryObject.name && queryObject.age) {
        res.end(`<h1>Hello ${queryObject.name}!</h1><p>You are ${queryObject.age} years old.</p>`);
    } else {
        res.end('<h1>Please provide name and age in the URL</h1><p>Example: ?name=John&age=25</p>');
    }
});
server.listen(3000, () => {
    console.log('Server running at http://localhost:3000');
});

#Write a Node JS program to demonstrate accessing static files using http webserver and client.


#server.js(file_name)
var fs=require('fs');
var http=require('http');
var url=require('url');
var ROOT_DIR='html';
http.createServer(function (req, res) {
    var urlObj=url.parse(req.url, true, false);
    fs.readFile(ROOT_DIR + urlObj.pathname, function (err, data) {
        if (err) {
            res.writeHead(404);
            res.end(JSON.stringify(err));
            return;
        }
        res.writeHead(200);
        res.end(data);
    }); 
}).listen(5000,function(){
    console.log("Server is running on port 5000...");
})

const { MongoClient } = require('mongodb');
const url = 'mongodb://localhost:27017';
const client = new MongoClient(url);
const dbName = 'myDatabase';
async function main() {
    await client.connect();
    console.log('Connected to MongoDB');
    const db = client.db(dbName);
    const collection = db.collection('students');

    await collection.insertOne({ name: "Noshank", age: 21, course: "CSE" });
    await collection.insertMany([
        { name: "Alice", age: 22, course: "IT" },
        { name: "Bob", age: 23, course: "ECE" }
    ]);
    console.log('Documents inserted');

    const allStudents = await collection.find().toArray();
    console.log('All Students:', allStudents);
    const oneStudent = await collection.findOne({ name: "Noshank" });
    console.log('Single Student:', oneStudent);

    await collection.updateOne(
        { name: "Noshank" }, 
        { $set: { age: 22 } }
    );
    console.log('Document updated');

    await collection.deleteOne({ name: "Bob" });
    console.log('Document deleted');
    await client.close();
}

main().catch(console.error);

#Perform Count, Limit, Sort, and Skip operations on the given collections using MongoDB.


const { MongoClient } = require('mongodb');
const url = 'mongodb://localhost:27017';
const client = new MongoClient(url);

async function runDemo() {
    try {
        await client.connect(); // 1️⃣ Connect to MongoDB
        const db = client.db('myDatabase');
        const collection = db.collection('students');

        // 2️⃣ Clear previous data and insert sample documents
        await collection.deleteMany({});
        const students = [
            { name: "Alice", age: 20, grade: "A" },
            { name: "Bob", age: 22, grade: "B" },
            { name: "Charlie", age: 21, grade: "A" },
            { name: "David", age: 23, grade: "C" }
        ];
        const insertResult = await collection.insertMany(students);
        console.log("Inserted Documents:", insertResult.insertedCount);

        // 3️⃣ Count documents
        const total = await collection.countDocuments();
        console.log("Total Documents:", total);

        // 4️⃣ Sort documents by name
        const sortedData = await collection.find().sort({ name: 1 }).toArray();
        console.log("Sorted Data:", sortedData);

        // 5️⃣ Limit to first 2 documents
        const limitedData = await collection.find().limit(2).toArray();
        console.log("Limited Data (First 2):", limitedData);

        // 6️⃣ Skip first 2 documents
        const skippedData = await collection.find().skip(2).toArray();
        console.log("Skipped Data (Skipping first 2):", skippedData);

    } catch (error) {
        console.error("Error:", error);
    } finally {
        await client.close(); // 7️⃣ Always close the connection
        console.log("Connection closed.");
    }
}

runDemo();
