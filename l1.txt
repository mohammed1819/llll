import numpy as np

# Create a 1D NumPy array
arr = np.array([10, 20, 30, 40, 50, 60, 70, 80])

print("Original Array:")
print(arr)

# -------------------------
# Slicing
# -------------------------
# Get elements from index 2 to 5
slice_arr = arr[2:6]
print("\nSliced Array (index 2 to 5):")
print(slice_arr)

# Get every second element
slice_step = arr[::2]
print("\nSliced Array (every 2nd element):")
print(slice_step)

# -------------------------
# Boolean Indexing
# -------------------------
# Get elements greater than 40
bool_index = arr[arr > 40]
print("\nElements greater than 40 (Boolean Indexing):")
print(bool_index)

# Get elements that are even
even_elements = arr[arr % 2 == 0]
print("\nEven Elements (Boolean Indexing):")
print(even_elements)











8a)
LabelEncoder converts categorical text into numeric codes.

For example:

"Finance" → 0

"HR" → 1

"IT" → 2

program:
import pandas as pd
from sklearn.preprocessing import LabelEncoder

# Sample DataFrame
data = {
    "Name": ["Alice", "Bob", "Charlie", "David", "Eva"],
    "Department": ["HR", "IT", "IT", "Finance", "HR"]
}

df = pd.DataFrame(data)
print("Original DataFrame:")
print(df)

# Initialize LabelEncoder
le = LabelEncoder()

# Apply Label Encoding on 'Department' column
df["Department_encoded"] = le.fit_transform(df["Department"])

print("\nDataFrame after Label Encoding:")
print(df)
